def calc(a,b,op) : 
    if op == "+" : 
        return a + b
    elif op == "-" : 
        return a - b
    elif op == "*" : 
        return a * b
    elif op == "/" : 
        return a / b
    
#--------------------------------------------------------------------------------

c = int(input("첫 번째 정수를 입력 >> "))
d = int(input("두 번째 정수를 입력 >> "))

print("{},{},{},{}이반환되었습니다.".format(calc(c,d,"+"),calc(c,d,"-"),calc(c,d,"*"),calc(c,d,"/")))

* Đầu tiên sẽ viết hàm checkPass : 
  cho tất cả các đơn vị ban đầu đều = 0 hết 
* Sau đó sẽ ra điều kiện : nếu độ dài của PASS >= 8 thì PASS sẽ + thêm 1 

* Vòng lặp for i in p : Dùng để duyệt từng kí tự trong mật khẩu đã nhập 

* Tương tự như điều kiện của hàm trên, nếu ký tự nào viết hoa(isupper()) ,viết thường (islower()) và có ký tự số (isdigit()) thì tất cả đơn vị tương ứng sẽ + thêm 1 

* Và điều kiện cuối cùng là mấu chốt : if PASS > 0 and u > 0 and l > 0 and d > 0
+ Điều kiện này bắt buộc mật khẩu nhập vào phải có độ dài kí tự >= 8,có chứ hoa, có chữ thường và có kí tự số thì mới có thể sử dụng đc 

#5 : 
def checkPass(p) :
    PASS = 0
    d = l = u = 0
    
    if len(p) >= 8 : PASS += 1
        
    for i in p:
        if i.isupper() : u += 1
        if i.islower() : l += 1
        if i.isdigit() : d += 1
            
    if PASS > 0 and u > 0 and l > 0 and d > 0 : print("사용할 수 있습니다")
    else         : print("사용할 수 없습니다")

#---------------------------------------------------------------------------------
checkPass("A1316885416")

def getMoneyText(amount) : 
    t = ["영","일","이","삼","사","오","육","칠","팔","구"]
    
    if amount // 100 > 0 : print(t[amount // 100],"백",end = "",sep = "")
    if amount % 100 // 10 > 0 : print(t[amount % 100 // 10],"십",end = "",sep = "")
    if amount % 10 > 0 : print(t[amount % 10],"원",sep = "")

amount = int(input("1000 이하의 금액을 입력 >>  "))
getMoneyText(amount)
