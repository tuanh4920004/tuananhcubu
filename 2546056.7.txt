# 상품의 가격이 100만원 미만이면 10% 할인이 적용된다.
#만약 상품의 가격이 100만원 이상이면 15%의 할인이 적용된다.
#그리고 100만원 이상 상품을 사면 사은품이 지급된다.
#100만원 미만이면 사은품은 없다

pr = int(input("상품 가격(만원 단위) : "))

pp = 0
if pr >= 100 :
    pr *= 0.85
    pp = 1
else         :
    pr *= 0.9
    
# 조건연산자 : 참일떼 if (조건문) else 居짓일떼

print("최종 가격은 {}원이며 사은품 수령 가능!".format(pr))  if pp == 1  else print("최종 가격은 {}만원".format(pr))

if pp == 1 : print("최종 가격은 {}원이며 사은품 수령 가능!".format(pr))
else : print("최종 가격은 {}만원".format(pr))
