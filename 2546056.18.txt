import random as r

def dice_game(): 
    user = int(input("인간 : 주사위값 = "))
    com  = r.randint(1,6)
    print("컴퓨터 : 주사위값 = ",com)
    
    if user > com : print("인간승리")
    elif user == com : print("인간 - 컴퓨터 비김")
    else :  print("컴퓨터승리")
#---------------------------------------------------------------------------        

while True :
    print("="*10,"dice_game() 호출","="*10)
    dice_game()
    print("="*10,"dice_game() 복귀","="*10)
    
    yn = input("중단할까요? Y/N  ")
    
    if yn == 'Y' or yn == 'y' : break

def testPrime(n) : 
    cnt = 0
    
    for i in range(1,33) : 
        if n % i == 0 : cnt += 1 
    
    if cnt == 2 : print(n, end = " ")
#--------------------------------------------------------------------------
for n in range(2,101) : 
    testPrime(n)

import turtle as t 

def draw_square(size,x = 30) : 
    size = 10 

    for n in range(1,x + 1) : 
        t.fd(size + (n * 5))
        t.left(90)
