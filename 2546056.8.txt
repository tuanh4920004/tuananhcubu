nhập cân nặng chiều cao và công thức tính BMI và các điều kiện in ra giá trị


a = float(input("무게 입력 >> "))
b = float(input("키 입력 >> "))
c = a/(b*b)
print("당신의 BMI: {}".format(c))
if c > 30 : print("비만입니다.")
elif c > 25 and c <=29.9 : print("과체중입니다.")
elif c > 20 and c <=24.9 : print("정상입니다.")
else : print("저체중입니다.")
