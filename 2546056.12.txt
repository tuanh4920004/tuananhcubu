Dùng để tạo mọi tổ hợp có thể giữa 2 nhóm dữ liệu

import random as r

adj = ["small","medium","large"]
nouns = ["apple","banana","grape"]

for x in adj : 
    for y in nouns:
        print(x, " = ",y)


for x in adj : 
    print(x," = ",r.choice(nouns))

import random as r

persons = ["Kim","Park","Lee","Choi"]
restaurants = ["Korea","America","French","Chinese"]
loc = ["서울","대전","대구","부산","제주도"]

for x in persons : 
    print(x,"에게는",r.choice(loc),"의",r.choice(restaurants),"식당을 추천해줄께")


#---------------------------------------------------------------------------------------
Code dùng để tính phép cộng : 1 + 1/2 + 1/3 + .... + 1/100

sum = 0 

for i in range(1,101) : 
    sum += 1.0 / i 

print("sum = ",sum)
#---------------------------------------------------------------------------------------
Thực hiện phép cộng và trừ phân số từ 1 đến 100

* Dòng điều kiện if i % 2 == 0 :     : Sẽ đặt điều kiện khi mẫu số chia hết cho 2 thì ở phía trc sẽ là dấu '-' 
                     sum -= 1.0 / i 
* Ngược lại nếu ko chia hết cho 2 thì phía trc sẽ đặt dấu '+'

sum = 0 

for i in range(1,101) : 
    if i % 2 == 0 :
        sum -= 1.0 / i
    else : sum += 1.0 / i

print("sum = ",sum)
