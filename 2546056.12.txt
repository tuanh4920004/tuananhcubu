import random as r

adj = ["small","medium","large"]
nouns = ["apple","banana","grape"]

for x in adj : 
    for y in nouns:
        print(x, " = ",y)


for x in adj : 
    print(x," = ",r.choice(nouns))

import random as r

persons = ["Kim","Park","Lee","Choi"]
restaurants = ["Korea","America","French","Chinese"]
loc = ["서울","대전","대구","부산","제주도"]

for x in persons : 
    print(x,"에게는",r.choice(loc),"의",r.choice(restaurants),"식당을 추천해줄께")

sum = 0 

for i in range(1,101) : 
    if i % 2 == 0 :
        sum -= 1.0 / i
    else : sum += 1.0 / i

print("sum = ",sum)
