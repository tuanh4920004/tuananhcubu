PI = 3.141592

#원의 반지름 
r = float(input("반지름 : "))
area = PI * r * r 
h = float(input(" 높이 : "))
vol = area * h 

print("반지름 : {}, 높이 : {} 일때, \n".format(r,h))
print("원의 넓이 : {}, 원기둥의 부피 : {}\n".format(area,vol))
