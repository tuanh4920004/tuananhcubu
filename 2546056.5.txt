Tạo 2 số ngẫu nhiên r tìm số lớn nhất và nhỏ nhất



import random as r

x = r.randint(1,10)
y = r.randint(1,10)

max = x if x>y else y 
min = x if x<y else y 
print("max=",max,"min=",min)

if x>y : max = x ; min = y
else:    max = y ; min = x
print("max=",max,"min=",min)
