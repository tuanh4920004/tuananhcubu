Vẽ sao 50 lần với vị trí ngẫu nhiên trên nền xanh và ngôi sao màu cam

import turtle as t
import random as r

t.home()
t.clear()

t.bgcolor("navy")
t.color("orange")

for i in range(50) : 
    x = r.randint(-250,250)
    y = r.randint(-250,250)
    t.penup()
    t.goto(x,y)
    t.pendown()
    
    t.begin_fill()
    for i in range(5): 
        t.forward(150)
        t.left(144)
    t.end_fill()
